var notificationTestMode sync.Once

type notificationTokens struct {
	expires time.Time
	err     error
	session string
	revoked atomic.Bool
}

func (v *notificationTokens) VerifyWithExpiry(raw string) (string, time.Time, error) {
	return raw, v.expires, v.err
}
func (v *notificationTokens) VerifyAuthorization(ctx context.Context, raw string) (string, string, string, time.Time, error) {
	return raw, "", v.session, v.expires, v.err
}
func (v *notificationTokens) AuthorizeSession(context.Context, string, string) error {
	if v.revoked.Load() {
		return apperrors.ErrUnauthorized
	}
	return nil
}

// The legacy path must remain supported when no durable-session verifier exists.
type notificationLegacyTokens struct{ expiry time.Time }

func (v notificationLegacyTokens) VerifyWithExpiry(raw string) (string, time.Time, error) {
	return raw, v.expiry, nil
}

type notificationFailedSubscriber struct{ err error }

func (s notificationFailedSubscriber) Subscribe(context.Context, string) (*goredis.PubSub, error) {
	return nil, s.err
}

type notificationBlockingSubscriber struct {
	started chan string
	release chan struct{}
}

func (s *notificationBlockingSubscriber) Subscribe(ctx context.Context, user string) (*goredis.PubSub, error) {
	s.started <- user
	select {
	case <-s.release:
		return nil, apperrors.ErrUnavailable
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}

type notificationLimiter struct {
	allowed bool
	err     error
}

func (l notificationLimiter) Allow(context.Context, string, int, time.Duration) (ratelimit.Result, error) {
	return ratelimit.Result{Allowed: l.allowed}, l.err
}

type notificationVisibility struct {
	usecase.Repository
	visible func(context.Context, string, string) (bool, error)
}

func (r notificationVisibility) Visible(ctx context.Context, user, id string) (bool, error) {
	return r.visible(ctx, user, id)
}

// The recorder is read only after the handler returns. Writes cross the channel
// as copied strings, so tests never race the response body or headers.
type notificationWriter struct {
	*httptest.ResponseRecorder
	writes    chan string
	deadlines []time.Time
	writeErr  error
}

func (w *notificationWriter) Write(data []byte) (int, error) {
	if w.writeErr != nil {
		return 0, w.writeErr
	}
	n, err := w.ResponseRecorder.Write(data)
	w.writes <- string(data)
	return n, err
}
func (w *notificationWriter) SetWriteDeadline(value time.Time) error {
	w.deadlines = append(w.deadlines, value)
	return nil
}

type notificationRequest struct {
	context *gin.Context
	writer  *notificationWriter
	cancel  context.CancelFunc
	done    chan struct{}
}

func newNotificationRequest(user, authorization, query string) *notificationRequest {
	notificationTestMode.Do(func() { gin.SetMode(gin.TestMode) })
	w := &notificationWriter{ResponseRecorder: httptest.NewRecorder(), writes: make(chan string, 64)}
	c, _ := gin.CreateTestContext(w)
	ctx, cancel := context.WithCancel(context.Background())
	c.Request = httptest.NewRequest(http.MethodGet, "/api/v1/notifications/events"+query, nil).WithContext(ctx)
	c.Request.Header.Set("Authorization", authorization)
	c.Set("authenticated_user_id", user)
	return &notificationRequest{context: c, writer: w, cancel: cancel, done: make(chan struct{})}
}

func (r *notificationRequest) start(t *testing.T, h *Handler) {
	t.Helper()
	go func() {
		defer close(r.done)
		h.Events(r.context)
	}()
	t.Cleanup(func() {
		r.cancel()
		select {
		case <-r.done:
		case <-time.After(3 * time.Second):
			t.Error("notification handler did not stop after request cancellation")
		}
	})
}

func (r *notificationRequest) waitDone(t *testing.T) {
	t.Helper()
	select {
	case <-r.done:
	case <-time.After(3 * time.Second):
		t.Fatal("notification stream did not finish")
	}
}

func (r *notificationRequest) nextWrite(t *testing.T) string {
	t.Helper()
	select {
	case value := <-r.writer.writes:
		return value
	case <-time.After(3 * time.Second):
		t.Fatal("notification stream did not produce its next frame")
		return ""
	}
}

func TestP2NotificationEventsHTTPContract(t *testing.T) {
	user := uuid.NewString()
	for _, tc := range []struct {
		name, authorization, query string
		tokens                     Verifier
		limiter                    Limiter
		subErr                     error
		status                     int
		body                       string
		retry                      string
	}{
		{name: "missing header", status: 401, body: "authentication required or token invalid"},
		{name: "credentials in query rejected", authorization: "Bearer " + user, query: "?token=private", status: 401, body: "authentication required or token invalid"},
		{name: "actor mismatch", authorization: "Bearer other", status: 401, body: "authentication required or token invalid"},
		{name: "invalid credential", authorization: "Bearer " + user, tokens: &notificationTokens{err: apperrors.ErrUnauthorized}, status: 401, body: "authentication required or token invalid"},
		{name: "rate limit", authorization: "Bearer " + user, limiter: notificationLimiter{}, status: 429, body: "too many notification connections", retry: "60"},
		{name: "limiter unavailable", authorization: "Bearer " + user, limiter: notificationLimiter{err: apperrors.ErrUnavailable}, status: 503, body: "service temporarily unavailable"},
		{name: "subscription unavailable", authorization: "Bearer " + user, subErr: apperrors.ErrUnavailable, status: 503, body: "service temporarily unavailable"},
		{name: "raw subscription error private", authorization: "Bearer " + user, subErr: errors.New("private redis address"), status: 500, body: "internal server error"},
	} {
		t.Run(tc.name, func(t *testing.T) {
			if tc.tokens == nil {
				tc.tokens = notificationLegacyTokens{expiry: time.Now().Add(time.Hour)}
			}
			h := NewHandler(nil, notificationFailedSubscriber{err: tc.subErr}, tc.tokens, tc.limiter, "p2-notifications")
			r := newNotificationRequest(user, tc.authorization, tc.query)
			defer r.cancel()
			h.Events(r.context)
			if r.writer.Code != tc.status || !strings.Contains(r.writer.Body.String(), tc.body) || r.writer.Header().Get("Retry-After") != tc.retry {
				t.Fatalf("status=%d body=%s retry=%q", r.writer.Code, r.writer.Body.String(), r.writer.Header().Get("Retry-After"))
			}
		})
	}
}

func TestP2NotificationSubscriptionFailureReleasesCapacity(t *testing.T) {
	user := uuid.NewString()
	h := NewHandler(nil, notificationFailedSubscriber{err: apperrors.ErrNotFound}, notificationLegacyTokens{expiry: time.Now().Add(time.Hour)}, nil, "p2-notifications")
	for i := 0; i < 300; i++ {
		r := newNotificationRequest(user, "Bearer "+user, "")
		h.Events(r.context)
		r.cancel()
		if r.writer.Code != 404 {
			t.Fatalf("request %d leaked stream capacity: status=%d", i, r.writer.Code)
		}
	}
}

func TestP2NotificationConnectionLimits(t *testing.T) {
	for _, tc := range []struct {
		name          string
		count, status int
		sameUser      bool
	}{{"per user", 4, 429, true}, {"process", 256, 503, false}} {
		t.Run(tc.name, func(t *testing.T) {
			bus := &notificationBlockingSubscriber{started: make(chan string, tc.count), release: make(chan struct{})}
			h := NewHandler(nil, bus, notificationLegacyTokens{expiry: time.Now().Add(time.Hour)}, nil, "p2-notifications")
			defer close(bus.release)
			user := uuid.NewString()
			requests := make([]*notificationRequest, 0, tc.count)
			for i := 0; i < tc.count; i++ {
				if !tc.sameUser {
					user = uuid.NewString()
				}
				r := newNotificationRequest(user, "Bearer "+user, "")
				r.start(t, h)
				requests = append(requests, r)
			}
			for i := 0; i < tc.count; i++ {
				select {
				case <-bus.started:
				case <-time.After(3 * time.Second):
					t.Fatal("capacity was not acquired")
				}
			}
			r := newNotificationRequest(user, "Bearer "+user, "")
			h.Events(r.context)
			r.cancel()
			if r.writer.Code != tc.status {
				t.Fatalf("limit status=%d, want %d", r.writer.Code, tc.status)
			}
			for _, request := range requests {
				request.cancel()
				request.waitDone(t)
			}
			// A new request reaches Subscribe after all previous slots were released.
			r = newNotificationRequest(user, "Bearer "+user, "")
			r.start(t, h)
			select {
			case <-bus.started:
			case <-time.After(3 * time.Second):
				t.Fatal("capacity was not released")
			}
			r.cancel()
			r.waitDone(t)
		})
	}
}

func notificationRedis(t *testing.T) *goredis.Client {
	t.Helper()
	address := os.Getenv("RECORDER_TEST_REDIS_ADDR")
	if address == "" {
		t.Skip("set RECORDER_TEST_REDIS_ADDR to the disposable audit Redis")
	}
	client := goredis.NewClient(&goredis.Options{Addr: address})
	t.Cleanup(func() { _ = client.Close() })
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	if err := client.Ping(ctx).Err(); err != nil {
		t.Fatal(err)
	}
	return client
}

func notificationPublish(t *testing.T, client *goredis.Client, channel, data string) {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	defer cancel()
	if err := client.Publish(ctx, channel, data).Err(); err != nil {
		t.Fatal(err)
	}
}

func notificationUnsubscribed(t *testing.T, client *goredis.Client, channel string) {
	t.Helper()
	deadline := time.Now().Add(time.Second)
	for time.Now().Before(deadline) {
		ctx, cancel := context.WithTimeout(context.Background(), time.Second)
		counts, err := client.PubSubNumSub(ctx, channel).Result()
		cancel()
		if err != nil {
			t.Fatal(err)
		}
		if counts[channel] == 0 {
			return
		}
		time.Sleep(time.Millisecond)
	}
	t.Fatal("Redis notification subscription remained open")
}

func notificationEnvelope(kind, id string) string {
	data, _ := json.Marshal(domain.Envelope{Version: 1, ID: id, Type: kind, Data: json.RawMessage(`{}`)})
	return string(data)
}

func TestP2NotificationEventDelivery(t *testing.T) {
	client := notificationRedis(t)
	namespace, user := "p2-notifications-"+uuid.NewString(), uuid.NewString()
	channel := namespace + ":notification:" + user
	visibleCalls := make(chan string, 4)
	repo := notificationVisibility{visible: func(_ context.Context, actor, id string) (bool, error) {
		visibleCalls <- actor + ":" + id
		return id == "visible", nil
	}}
	tokens := &notificationTokens{expires: time.Now().Add(time.Hour), session: "session"}
	h := NewHandler(usecase.NewService(repo, nil), redisinfra.NewNotificationBus(client, namespace), tokens, nil, namespace)
	r := newNotificationRequest(user, "Bearer "+user, "")
	r.start(t, h)
	if frame := r.nextWrite(t); frame != ": connected\n\n" {
		t.Fatalf("connected frame=%q", frame)
	}
	for _, event := range []string{notificationEnvelope("unrelated", "ignored"), notificationEnvelope("notification.created", "hidden"), notificationEnvelope("notification.created", "visible"), notificationEnvelope("notification.read", "read")} {
		notificationPublish(t, client, channel, event)
	}
	if frame := r.nextWrite(t); frame != "data: "+notificationEnvelope("notification.created", "visible")+"\n\n" {
		t.Fatalf("visible frame=%q", frame)
	}
	if frame := r.nextWrite(t); frame != "data: "+notificationEnvelope("notification.read", "read")+"\n\n" {
		t.Fatalf("read frame=%q", frame)
	}
	for _, id := range []string{"hidden", "visible"} {
		select {
		case call := <-visibleCalls:
			if call != user+":"+id {
				t.Fatalf("visibility authorization=%q", call)
			}
		case <-time.After(time.Second):
			t.Fatal("missing visibility authorization")
		}
	}
	r.cancel()
	r.waitDone(t)
	if r.writer.Code != 200 || r.writer.Header().Get("Content-Type") != "text/event-stream" || r.writer.Header().Get("Cache-Control") != "no-store" || r.writer.Header().Get("X-Accel-Buffering") != "no" {
		t.Fatalf("SSE response status=%d headers=%v", r.writer.Code, r.writer.Header())
	}
	if len(r.writer.deadlines) != 4 || !r.writer.deadlines[len(r.writer.deadlines)-1].IsZero() {
		t.Fatalf("write deadlines not cleared: %v", r.writer.deadlines)
	}
	notificationUnsubscribed(t, client, channel)
}

func TestP2NotificationStreamCleanup(t *testing.T) {
	client := notificationRedis(t)
	for _, kind := range []string{"cancel", "expiry", "drain", "revoked", "malformed", "oversize", "visibility error", "write error"} {
		t.Run(kind, func(t *testing.T) {
			namespace, user := "p2-notifications-"+uuid.NewString(), uuid.NewString()
			channel := namespace + ":notification:" + user
			tokens := &notificationTokens{expires: time.Now().Add(time.Hour), session: "session"}
			if kind == "expiry" {
				tokens.expires = time.Now().Add(time.Second)
			}
			repo := notificationVisibility{visible: func(context.Context, string, string) (bool, error) {
				return false, errors.New("visibility store unavailable")
			}}
			h := NewHandler(usecase.NewService(repo, nil), redisinfra.NewNotificationBus(client, namespace), tokens, nil, namespace)
			r := newNotificationRequest(user, "Bearer "+user, "")
			if kind == "write error" {
				r.writer.writeErr = errors.New("client disconnected")
			}
			r.start(t, h)
			if kind != "write error" {
				if frame := r.nextWrite(t); frame != ": connected\n\n" {
					t.Fatalf("connected frame=%q", frame)
				}
			}
			switch kind {
			case "cancel":
				r.cancel()
			case "drain":
				var wg sync.WaitGroup
				for range 32 {
					wg.Add(1)
					go func() { defer wg.Done(); h.BeginDrain() }()
				}
				wg.Wait()
			case "revoked":
				tokens.revoked.Store(true)
				notificationPublish(t, client, channel, notificationEnvelope("notification.read", "read"))
			case "malformed":
				notificationPublish(t, client, channel, "{")
			case "oversize":
				notificationPublish(t, client, channel, strings.Repeat("x", 8193))
			case "visibility error":
				notificationPublish(t, client, channel, notificationEnvelope("notification.created", "private"))
			}
			r.waitDone(t)
			if len(r.writer.deadlines) == 0 || !r.writer.deadlines[len(r.writer.deadlines)-1].IsZero() {
				t.Fatal("write deadline was not cleared")
			}
			notificationUnsubscribed(t, client, channel)
		})
	}
}

	h := NewHandler(nil, nil, nil, nil, "audit")

	for b.Loop() {

	}
}
