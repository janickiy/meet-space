package httpresponse

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"net/http/httptest"
	"testing"

	"github.com/gin-gonic/gin"
	"github.com/janickiy/meet-space/internal/domain/apperrors"
)

func TestP2HTTPErrorContracts(t *testing.T) {
	gin.SetMode(gin.TestMode)
	cases := []struct {
		kind    error
		status  int
		message string
	}{
		{apperrors.ErrInvalidInput, 422, "invalid input"},
		{apperrors.ErrUnauthorized, 401, "authentication required or token invalid"},
		{apperrors.ErrForbidden, 403, "access denied"},
		{apperrors.ErrNotFound, 404, "not found"},
		{apperrors.ErrConflict, 409, "conflict"},
		{apperrors.ErrRateLimited, 429, "rate limit exceeded"},
		{apperrors.ErrUnavailable, 503, "service temporarily unavailable"},
		{apperrors.ErrTimeout, 504, "operation timed out"},
		{apperrors.ErrInternal, 500, "internal server error"},
	}
	assert := func(t *testing.T, err error, status int, message string) {
		t.Helper()
		w := httptest.NewRecorder()
		c, _ := gin.CreateTestContext(w)
		Fail(c, err)
		if w.Code != status || !c.IsAborted() {
			t.Fatalf("status=%d aborted=%v", w.Code, c.IsAborted())
		}
		if got := w.Header().Get("Content-Type"); got != "application/json; charset=utf-8" {
			t.Fatalf("content type %q", got)
		}
		var body map[string]any
		if json.Unmarshal(w.Body.Bytes(), &body) != nil || len(body) != 2 || body["status"] != "failed" || body["message"] != message {
			t.Fatalf("unexpected wire body %s", w.Body.String())
		}
	}
	for _, tc := range cases {
		t.Run(tc.kind.Error(), func(t *testing.T) {
			assert(t, tc.kind, tc.status, tc.message)
			message := "safe operation explanation"
			if tc.status == 500 {
				message = "internal server error"
			}
			err := apperrors.Wrap(tc.kind, errors.New("private SQL/credential details"), "safe operation explanation")
			assert(t, fmt.Errorf("private boundary: %w", err), tc.status, message)
		})
	}
	for _, err := range []error{nil, context.DeadlineExceeded, context.Canceled, errors.New("private storage details")} {
		assert(t, err, 500, "internal server error")
	}
	assert(t, apperrors.Wrap(apperrors.ErrForbidden, apperrors.ErrNotFound, "resource denied"), 403, "resource denied")
}
