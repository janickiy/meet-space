type p2MediaControl func(context.Context, domain.Session, time.Time, domain.Envelope) error

func (f p2MediaControl) Handle(ctx context.Context, session domain.Session, expiry time.Time, event domain.Envelope) error {
	return f(ctx, session, expiry, event)
}

func TestP2WebSocketMediaErrors(t *testing.T) {
	cases := []struct {
		err  error
		code string
	}{
		{media.ErrUnavailable, "media_unavailable"}, {media.ErrInvalid, "invalid_media_signal"},
		{media.ErrUnauthorized, "media_unauthorized"}, {media.ErrOwnership, "media_ownership_lost"},
		{media.ErrLimit, "media_limit_exceeded"}, {media.ErrPeerNotFound, "media_peer_not_found"},
		{media.ErrNegotiation, "media_negotiation_conflict"}, {media.ErrScreenConflict, "screen_sharing_conflict"},
		{media.ErrPolicy, "media_policy_blocked"}, {errors.New("private diagnostic"), "media_unavailable"},
	}
	for _, tc := range cases {
		t.Run(tc.code, func(t *testing.T) {
			session := domain.Session{ConferenceID: "room", UserID: "actor", ConnectionID: "connection"}
			event := domain.Event("media.join", "room", nil)
			var operation context.Context
			h := &Handler{cfg: config.RealtimeConfig{QueueSize: 1}}
			h.media = p2MediaControl(func(ctx context.Context, got domain.Session, expiry time.Time, request domain.Envelope) error {
				operation = ctx
				if got != session || request.ID != event.ID {
					t.Fatal("trusted request identity changed")
				}
				if deadline, ok := ctx.Deadline(); !ok || time.Until(deadline) > 10*time.Second {
					t.Fatal("media operation is not bounded")
				}
				return fmt.Errorf("private transport context: %w", tc.err)
			})
			c := newClient(nil, h, session, time.Time{})
			c.handleMedia(event)
			if operation.Err() != context.Canceled {
				t.Fatalf("operation context not released: %v", operation.Err())
			}
			select {
			case got := <-c.out:
				var body map[string]any
				if json.Unmarshal(got.Data, &body) != nil || len(body) != 1 || body["code"] != tc.code || got.Type != "error" || got.ReplyTo != event.ID || got.ConferenceID != session.ConferenceID || got.Version != 1 {
					t.Fatalf("invalid error envelope: %+v %s", got, got.Data)
				}
			default:
				t.Fatal("missing error reply")
			}
		})
	}
}

func TestP2WebSocketReplyCorrelationBound(t *testing.T) {
	h := &Handler{cfg: config.RealtimeConfig{QueueSize: 1}}
	c := newClient(nil, h, domain.Session{ConferenceID: "room"}, time.Time{})
	for _, reply := range []string{"", strings.Repeat("a", 36), strings.Repeat("a", 37)} {
		c.failure(reply, "invalid_message")
		got := <-c.out
		want := reply
		if len(reply) > 36 {
			want = ""
		}
		if got.ReplyTo != want {
			t.Fatalf("reply=%q want=%q", got.ReplyTo, want)
		}
	}
	c.handleMedia(domain.Event("media.join", "room", nil))
	if got := <-c.out; string(got.Data) != `{"code":"media_unavailable"}` {
		t.Fatalf("nil media response=%s", got.Data)
	}
}
