func p2JoinBinding(conferenceID string) media.Binding {
	return media.Binding{ConferenceID: conferenceID, ParticipantID: uuid.NewString(), SessionID: uuid.NewString(), ConnectionID: uuid.NewString()}
}

func p2JoinManager(t testing.TB) *Manager {
	t.Helper()
	m, err := NewManager(Options{MaxRooms: 1, MaxPeers: 2, NegotiationTimeout: time.Minute})
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() {
		ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
		defer cancel()
		if err := m.Shutdown(ctx); err != nil {
			t.Errorf("shutdown: %v", err)
		}
	})
	return m
}

// Each rejected Join has already allocated a real PeerConnection. Rejections
// must release it without disturbing registered peers or their room policies.
func TestP2JoinRejectionsRetainRegisteredPeers(t *testing.T) {
	for _, name := range []string{"max_rooms", "max_peers", "kicked"} {
		t.Run(name, func(t *testing.T) {
			baseline := runtime.NumGoroutine()
			m := p2JoinManager(t)
			conferenceID := uuid.NewString()
			accepted := p2JoinBinding(conferenceID)
			view, err := m.Join(context.Background(), accepted)
			if err != nil {
				t.Fatal(err)
			}
			rejected := p2JoinBinding(conferenceID)
			wantErr := media.ErrLimit
			switch name {
			case "max_rooms":
				rejected.ConferenceID = uuid.NewString()
			case "max_peers":
				if _, err := m.Join(context.Background(), p2JoinBinding(conferenceID)); err != nil {
					t.Fatal(err)
				}
			case "kicked":
				wantErr = media.ErrPolicy
				if err := m.SetPolicy(context.Background(), conferenceID, rejected.ParticipantID, media.ParticipantPolicy{Version: 1, Kicked: true}); err != nil {
					t.Fatal(err)
				}
			}
			before := m.Snapshot()
			for i := 0; i < 30; i++ {
				got, err := m.Join(context.Background(), rejected)
				if !errors.Is(err, wantErr) || got != (media.PeerView{}) {
					t.Fatalf("rejection %d: view=%+v error=%v, want empty view and %v", i, got, err, wantErr)
				}
				if after := m.Snapshot(); after != before {
					t.Fatalf("rejection %d changed registered state: before=%+v after=%+v", i, before, after)
				}
			}
			if got, err := m.Join(context.Background(), accepted); err != nil || got != view {
				t.Fatalf("registered connection lost idempotent Join: got=%+v err=%v", got, err)
			}
			if name == "kicked" {
				policy, ok := m.ParticipantPolicy(conferenceID, rejected.ParticipantID)
				if !ok || !policy.Kicked || policy.Version != 1 {
					t.Fatalf("rejected peer changed admission policy: %+v, exists=%v", policy, ok)
				}
			}
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			defer cancel()
			if err := m.Shutdown(ctx); err != nil {
				t.Fatal(err)
			}
			if got := m.Snapshot(); got.Peers != 0 || got.Rooms != 0 {
				t.Fatalf("shutdown retained registry entries: %+v", got)
			}
			deadline := time.Now().Add(3 * time.Second)
			for runtime.NumGoroutine() > baseline+3 && time.Now().Before(deadline) {
				time.Sleep(10 * time.Millisecond)
			}
			if got := runtime.NumGoroutine(); got > baseline+3 {
				t.Fatalf("rejected PeerConnections retained goroutines: before=%d after=%d", baseline, got)
			}
		})
	}
}

// An unregistered peer has no room or tracks, but already owns its transport,
// context and any started work. Concurrent close must join that work once.
func TestP2UnregisteredPeerClose(t *testing.T) {
	pc, err := pion.NewPeerConnection(pion.Configuration{})
	if err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithCancel(context.Background())
	p := &peer{pc: pc, ctx: ctx, cancel: cancel}
	t.Cleanup(p.close)
	joined := make(chan struct{})
	if !p.start(func() { <-ctx.Done(); close(joined) }) {
		t.Fatal("unregistered peer refused its owned work")
	}
	var closes sync.WaitGroup
	for i := 0; i < 8; i++ {
		closes.Add(1)
		go func() { defer closes.Done(); p.close() }()
	}
	done := make(chan struct{})
	go func() { closes.Wait(); close(done) }()
	select {
	case <-done:
	case <-time.After(3 * time.Second):
		t.Fatal("concurrent close did not finish")
	}
	if pc.ConnectionState() != pion.PeerConnectionStateClosed || !errors.Is(ctx.Err(), context.Canceled) {
		t.Fatalf("owned transport/context still active: state=%s err=%v", pc.ConnectionState(), ctx.Err())
	}
	select {
	case <-joined:
	default:
		t.Fatal("close returned before its owned work completed")
	}
	lateWork := make(chan struct{}, 1)
	if p.start(func() { lateWork <- struct{}{} }) {
		t.Fatal("closed peer accepted new work")
	}
	select {
	case <-lateWork:
		t.Fatal("work ran after peer close")
	default:
	}
}
