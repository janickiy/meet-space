package personal

import (
	"context"
	domain "github.com/janickiy/meet-space/internal/domain/realtime"
	"testing"
)

type preferencesEventBus struct{ users []string }

func (b *preferencesEventBus) Publish(_ context.Context, user string, _ domain.Envelope) error {
	b.users = append(b.users, user)
	return nil
}

func TestGroupPreferencesEventIsPersonal(t *testing.T) {
	for _, tc := range []struct {
		kind, typ, target string
		allowed           bool
	}{
		{"conversation.preferences.updated", "group", "actor", true},
		{"conversation.preferences.updated", "direct", "actor", true},
		{"conversation.preferences.updated", "group", "other", false},
		{"conversation.history.cleared", "group", "actor", false},
		{"conversation.hidden", "group", "actor", false},
		{"conversation.preferences.updated", "conference", "actor", false},
	} {
		bus := &preferencesEventBus{}
		e := &Events{Bus: bus}
		err := e.PublishUser(context.Background(), "actor", tc.kind, map[string]any{"conversationId": "94563614-53b0-4820-a7b6-ea5710655cd4", "type": tc.typ, "userId": tc.target})
		if (err == nil) != tc.allowed {
			t.Fatalf("%+v: err=%v", tc, err)
		}
		if tc.allowed {
			if len(bus.users) != 1 || bus.users[0] != "actor" {
				t.Fatalf("unexpected recipients: %v", bus.users)
			}
		} else if len(bus.users) != 0 {
			t.Fatal("invalid private event published")
		}
	}
}
