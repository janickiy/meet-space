package postgres_test

import (
	"context"
	"encoding/json"
	"github.com/google/uuid"
	ri "github.com/janickiy/meet-space/internal/infrastructure/redis"
	gr "github.com/redis/go-redis/v9"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"
)

func TestP1AccountPresenceBatches(t *testing.T) {
	addr := os.Getenv("RECORDER_P1_REDIS_ADDR")
	if !strings.HasPrefix(addr, "127.0.0.1:") {
		t.Skip("isolated loopback Redis required")
	}
	client, err := ri.NewClient(context.Background(), addr, "", 0)
	if err != nil {
		t.Fatal(err)
	}
	defer client.Close()
	ns := "p1-online-" + uuid.NewString()
	hook := &p1RedisHook{}
	client.AddHook(hook)
	presence := ri.NewUserPresence(client, ns, 5*time.Second)
	ids := make([]string, 100)
	for i := range ids {
		ids[i] = uuid.NewString()
		if _, err = presence.Touch(context.Background(), ids[i], "fixture"); err != nil {
			t.Fatal(err)
		}
	}
	defer func() {
		keys, _ := client.Keys(context.Background(), ns+":user-presence:*").Result()
		if len(keys) > 0 {
			_ = client.Del(context.Background(), keys...).Err()
		}
	}()
	rows := []map[string]any{}
	for _, size := range []int{1, 20, 100} {
		if _, err = presence.Online(context.Background(), ids[:size]); err != nil {
			t.Fatal(err)
		}
		hook.batches.Store(0)
		hook.commands.Store(0)
		start := time.Now()
		v, err := presence.Online(context.Background(), ids[:size])
		elapsed := time.Since(start)
		if err != nil || len(v) != size {
			t.Fatal("presence mismatch", err)
		}
		for _, online := range v {
			if !online {
				t.Fatal("active account offline")
			}
		}
		rows = append(rows, map[string]any{"accounts": size, "batches": hook.batches.Load(), "commands": hook.commands.Load(), "elapsedMicroseconds": float64(elapsed) / float64(time.Microsecond)})
	}
	key := ns + ":user-presence:" + ids[0]
	before, err := client.PTTL(context.Background(), key).Result()
	if err != nil {
		t.Fatal(err)
	}
	time.Sleep(20 * time.Millisecond)
	if _, err = presence.Online(context.Background(), ids); err != nil {
		t.Fatal(err)
	}
	after, err := client.PTTL(context.Background(), key).Result()
	if err != nil || after >= before {
		t.Fatal("presence read extended TTL")
	}
	path := filepath.Join(os.Getenv("RECORDER_P1_PROFILE_DIR"), "account-presence.json")
	data, _ := json.MarshalIndent(rows, "", "  ")
	if err = os.WriteFile(path, data, 0600); err != nil {
		t.Fatal(err)
	}
	t.Log(string(data))
	// Keep the SDK type in the source contract when hooks are updated.
	var _ gr.Hook = hook
}
