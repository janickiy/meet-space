func groupFixture(t testing.TB) (*gorm.DB, []string) {
	t.Helper()
	db := stageOneDatabase(t)
	ids := make([]string, 100)
	for i := range ids {
		ids[i] = p1ID("group-user", i)
	}
	if err := db.Exec(`INSERT INTO users(id,email,password_hash,display_name) SELECT md5('group-user'||i)::uuid,'group-'||i||'@example.invalid','fixture-not-a-password','Participant '||i FROM generate_series(0,99) i`).Error; err != nil {
		t.Fatal(err)
	}
	return db, ids
}

func BenchmarkP1GroupCreation(b *testing.B) {
	if os.Getenv("RECORDER_P1_GROUP_PERF") != "true" {
		b.Skip("isolated audit database required")
	}
	for _, size := range []int{2, 20, 100} {
		b.Run(fmt.Sprintf("members_%d", size), func(b *testing.B) {
			db, ids := groupFixture(b)
			repo := pg.NewPersonalRepository(db)
			b.ReportAllocs()
			b.ResetTimer()
			for b.Loop() {
				item, created, err := repo.CreateGroup(context.Background(), ids[0], personal.CreateGroupRequest{ClientRequestID: uuid.NewString(), Name: "Benchmark fixture", MemberIDs: ids[1:size]})
				if err != nil || !created || item.ID == "" {
					b.Fatalf("create: %v", err)
				}
			}
		})
	}
}

func TestP1GroupPerformance(t *testing.T) {
	if os.Getenv("RECORDER_P1_GROUP_PERF") != "true" {
		t.Skip("opt-in group workload")
	}
	db, ids := groupFixture(t)
	repo := pg.NewPersonalRepository(db)
	pool, _ := db.DB()
	var queries atomic.Int64
	hook := func(*gorm.DB) { queries.Add(1) }
	for key, register := range map[string]func(string, func(*gorm.DB)) error{"raw": db.Callback().Raw().After("gorm:raw").Register, "query": db.Callback().Query().After("gorm:query").Register, "row": db.Callback().Row().After("gorm:row").Register, "create": db.Callback().Create().After("gorm:create").Register} {
		if err := register("p1-group-"+key, hook); err != nil {
			t.Fatal(err)
		}
	}
	reports := []map[string]any{}
	for _, size := range []int{2, 20, 100} {
		create := func() (personal.Conversation, error) {
			v, created, err := repo.CreateGroup(context.Background(), ids[0], personal.CreateGroupRequest{ClientRequestID: uuid.NewString(), Name: "Performance fixture", MemberIDs: ids[1:size]})
			if err == nil && !created {
				return v, fmt.Errorf("group was not created")
			}
			return v, err
		}
		for i := 0; i < 3; i++ {
			if _, err := create(); err != nil {
				t.Fatal(err)
			}
		}
		runtime.GC()
		var before, after runtime.MemStats
		runtime.ReadMemStats(&before)
		queries.Store(0)
		times := make([]float64, 30)
		var last personal.Conversation
		for i := range times {
			start := time.Now()
			var err error
			last, err = create()
			times[i] = float64(time.Since(start)) / float64(time.Millisecond)
			if err != nil {
				t.Fatal(err)
			}
		}
		runtime.ReadMemStats(&after)
		count := queries.Load()
		sort.Float64s(times)
		people, err := repo.GroupMembers(context.Background(), ids[0], last.ID)
		if err != nil || len(people) != size {
			t.Fatalf("member count %d: %v", len(people), err)
		}
		owners := 0
		for _, person := range people {
			if person.Role == personal.Owner {
				owners++
			}
		}
		if owners != 1 {
			t.Fatal("owner count differs")
		}
		reports = append(reports, map[string]any{"members": size, "samples": len(times), "queriesPerCreate": count / int64(len(times)), "p50ms": times[14], "p95ms": times[28], "p99ms": times[29], "bytesPerOp": (after.TotalAlloc - before.TotalAlloc) / uint64(len(times)), "allocsPerOp": (after.Mallocs - before.Mallocs) / uint64(len(times)), "membershipAndSingleOwnerVerified": true})
	}
	group, _, err := repo.CreateGroup(context.Background(), ids[0], personal.CreateGroupRequest{ClientRequestID: uuid.NewString(), Name: "Concurrent readers", MemberIDs: ids[1:20]})
	if err != nil {
		t.Fatal(err)
	}
	var failed atomic.Int64
	var wg sync.WaitGroup
	before := pool.Stats()
	start := time.Now()
	for i := 0; i < 32; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for j := 0; j < 5; j++ {
				if _, err := repo.GroupMembers(context.Background(), ids[0], group.ID); err != nil {
					failed.Add(1)
				}
			}
		}()
	}
	wg.Wait()
	after := pool.Stats()
	reports = append(reports, map[string]any{"workload": "32 concurrent readers x 5 group member lists", "elapsedMs": float64(time.Since(start)) / float64(time.Millisecond), "poolWaitCount": after.WaitCount - before.WaitCount, "poolWaitMs": float64(after.WaitDuration-before.WaitDuration) / float64(time.Millisecond), "errors": failed.Load(), "poolAfter": after})
	if failed.Load() != 0 {
		t.Fatal("group read errors")
	}
	data, _ := json.MarshalIndent(reports, "", "  ")
	dir := os.Getenv("RECORDER_P1_DB_PROFILE_DIR")
	if err := os.MkdirAll(dir, 0700); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(dir, "group-performance.json"), data, 0600); err != nil {
		t.Fatal(err)
	}
	t.Log(string(data))
}

func TestP1GroupCreationContracts(t *testing.T) {
	if os.Getenv("RECORDER_P1_GROUP_PERF") != "true" {
		t.Skip("isolated audit database required")
	}
	db, ids := groupFixture(t)
	repo := pg.NewPersonalRepository(db)
	req := personal.CreateGroupRequest{ClientRequestID: uuid.NewString(), Name: "Contract fixture", MemberIDs: ids[1:100]}
	first, created, err := repo.CreateGroup(context.Background(), ids[0], req)
	if err != nil || !created {
		t.Fatalf("create: %v", err)
	}
	second, created, err := repo.CreateGroup(context.Background(), ids[0], req)
	if err != nil || created || first.ID != second.ID {
		t.Fatal("retry must return original group")
	}
	people, err := repo.GroupMembers(context.Background(), ids[0], first.ID)
	if err != nil || len(people) != 100 {
		t.Fatal("member list differs")
	}
	owners := 0
	for _, member := range people {
		if member.ID == ids[0] {
			if member.Role != personal.Owner {
				t.Fatal("creator must be owner")
			}
			owners++
		} else if member.Role != personal.MemberRole {
			t.Fatal("invited account must be member")
		}
	}
	if owners != 1 {
		t.Fatal("exactly one owner required")
	}
	req.ClientRequestID = uuid.NewString()
	// Copy before appending: mutating ids here would invalidate the later
	// rollback fixture and reject it before the injected INSERT trigger runs.
	req.MemberIDs = append(append([]string(nil), ids[1:99]...), uuid.NewString())
	if _, _, err = repo.CreateGroup(context.Background(), ids[0], req); err == nil {
		t.Fatal("unregistered account accepted")
	}
	req.ClientRequestID = uuid.NewString()
	req.MemberIDs = append(ids[1:], uuid.NewString())
	if _, _, err = repo.CreateGroup(context.Background(), ids[0], req); err == nil {
		t.Fatal("oversized group accepted")
	}
	var groups int64
	if err = db.Table("conversations").Where("type='group'").Count(&groups).Error; err != nil || groups != 1 {
		t.Fatal("failed creation must leave no group")
	}
	// A failed batch must roll back both group metadata and all memberships.
	if err = db.Exec(`CREATE FUNCTION p1_reject_member() RETURNS trigger LANGUAGE plpgsql AS $$ BEGIN IF NEW.user_id='` + ids[50] + `' THEN RAISE EXCEPTION 'audit injected membership failure'; END IF; RETURN NEW; END $$; CREATE TRIGGER p1_reject_member BEFORE INSERT ON conversation_members FOR EACH ROW EXECUTE FUNCTION p1_reject_member()`).Error; err != nil {
		t.Fatal(err)
	}
	req.ClientRequestID = uuid.NewString()
	req.MemberIDs = ids[1:100]
	if _, _, err = repo.CreateGroup(context.Background(), ids[0], req); err == nil {
		t.Fatal("injected membership failure ignored")
	} else if !strings.Contains(err.Error(), "audit injected membership failure") {
		t.Fatalf("expected INSERT trigger failure, got: %v", err)
	}
	if err = db.Table("conversations").Where("type='group'").Count(&groups).Error; err != nil || groups != 1 {
		t.Fatal("failed transaction left metadata")
	}
	var members int64
	if err = db.Table("conversation_members").Count(&members).Error; err != nil || members != 100 {
		t.Fatal("failed transaction left members")
	}
}
