package postgres_test

import (
	"context"
	"errors"
	"github.com/google/uuid"
	"github.com/janickiy/meet-space/internal/domain/apperrors"
	"github.com/janickiy/meet-space/internal/domain/chat"
	"github.com/janickiy/meet-space/internal/domain/personal"
	pg "github.com/janickiy/meet-space/internal/infrastructure/postgres"
	"strings"
	"testing"
)

func TestGroupPreferencesMembershipAndDelivery(t *testing.T) {
	db, ids := groupFixture(t)
	repo := pg.NewPersonalRepository(db)
	ctx := context.Background()
	group, _, err := repo.CreateGroup(ctx, ids[0], personal.CreateGroupRequest{ClientRequestID: uuid.NewString(), Name: "Group preference fixture", MemberIDs: ids[1:3]})
	if err != nil {
		t.Fatal(err)
	}
	for _, enabled := range []bool{false, false, true, false} {
		item, err := repo.SetConversationPreferences(ctx, ids[1], group.ID, enabled)
		if err != nil || item.Type != "group" || item.MyRole != personal.MemberRole || item.NotificationsEnabled != enabled {
			t.Fatalf("ordinary member setting %v: %+v %v", enabled, item, err)
		}
		persisted, err := pg.NewPersonalRepository(db).Get(ctx, ids[1], group.ID)
		if err != nil || persisted.NotificationsEnabled != enabled {
			t.Fatal("preference not persisted")
		}
		for _, other := range []string{ids[0], ids[2]} {
			view, err := repo.Get(ctx, other, group.ID)
			if err != nil || !view.NotificationsEnabled {
				t.Fatal("setting affected another member")
			}
		}
	}
	if _, err = repo.SetConversationPreferences(ctx, ids[3], group.ID, false); !errors.Is(err, apperrors.ErrForbidden) {
		t.Fatalf("nonmember allowed: %v", err)
	}
	message, _, err := pg.NewDirectChatRepository(db).Send(ctx, ids[0], group.ID, chat.SendRequest{ClientRequestID: uuid.NewString(), Text: "Muted messages remain available"}, strings.Repeat("a", 64))
	if err != nil {
		t.Fatal(err)
	}
	delivered := false
	allowed, err := repo.WithConversationMessageDelivery(ctx, group.ID, ids[1], message.ID, func(enabled bool, _ int64) error {
		delivered = true
		if enabled {
			t.Fatal("muted group notification enabled")
		}
		return nil
	})
	if err != nil || !allowed || !delivered {
		t.Fatalf("muted messages suppressed: %v %v", allowed, err)
	}
	allowed, err = repo.WithConversationActionDelivery(ctx, group.ID, ids[1], "conversation.preferences.updated", func(item *personal.Conversation) error {
		if item == nil || item.Type != "group" || item.NotificationsEnabled {
			t.Fatal("invalid authoritative preferences")
		}
		return nil
	})
	if err != nil || !allowed {
		t.Fatalf("group preferences unavailable: %v %v", allowed, err)
	}
	allowed, err = repo.WithConversationActionDelivery(ctx, group.ID, ids[1], "conversation.history.cleared", func(*personal.Conversation) error { t.Fatal("group history clear event allowed"); return nil })
	if err != nil || allowed {
		t.Fatalf("unsupported group action: %v %v", allowed, err)
	}
	if err = repo.LeaveGroup(ctx, ids[1], group.ID); err != nil {
		t.Fatal(err)
	}
	if _, err = repo.SetConversationPreferences(ctx, ids[1], group.ID, true); !errors.Is(err, apperrors.ErrForbidden) {
		t.Fatalf("former member allowed: %v", err)
	}
	allowed, err = repo.WithConversationActionDelivery(ctx, group.ID, ids[1], "conversation.preferences.updated", func(*personal.Conversation) error { t.Fatal("former member action delivered"); return nil })
	if err != nil || allowed {
		t.Fatalf("former member delivery: %v %v", allowed, err)
	}
	direct, _, err := repo.GetOrCreate(ctx, ids[0], ids[3])
	if err != nil {
		t.Fatal(err)
	}
	item, err := repo.SetConversationPreferences(ctx, ids[3], direct.ID, false)
	if err != nil || item.Type != "direct" || item.NotificationsEnabled {
		t.Fatalf("direct setting regressed: %+v %v", item, err)
	}
	if _, err = repo.DeleteGroup(ctx, ids[0], group.ID); err != nil {
		t.Fatal(err)
	}
	if _, err = repo.SetConversationPreferences(ctx, ids[0], group.ID, false); !errors.Is(err, apperrors.ErrForbidden) {
		t.Fatalf("deleted group allowed: %v", err)
	}
}
