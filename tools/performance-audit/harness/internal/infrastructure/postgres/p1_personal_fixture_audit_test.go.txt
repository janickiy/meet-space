package postgres_test

import (
	"gorm.io/gorm"
	"testing"
)

func seedP1Personal(t *testing.T, db *gorm.DB) {
	t.Helper()
	queries := []string{
		`INSERT INTO conversations(id,type,name,description,created_by,group_create_request_id,group_request_fingerprint) SELECT md5('audit-group'||i)::uuid,'group','Audit group '||i,'Synthetic fixture',md5('user1')::uuid,md5('audit-group-request'||i)::uuid,repeat('a',64) FROM generate_series(1,1000)i`,
		`INSERT INTO conversation_members(conversation_id,user_id,role) SELECT c.id,u.id,CASE WHEN u.id=md5('user1')::uuid THEN 'owner' ELSE 'member' END FROM conversations c CROSS JOIN users u WHERE c.type='group'`,
		`INSERT INTO conversations(id,user_low_id,user_high_id) SELECT md5('audit-direct'||i)::uuid,LEAST(md5('user1')::uuid,md5('user'||i)::uuid),GREATEST(md5('user1')::uuid,md5('user'||i)::uuid) FROM generate_series(2,20)i`,
		`INSERT INTO conversation_members(conversation_id,user_id) SELECT c.id,x.user_id FROM conversations c CROSS JOIN LATERAL (VALUES (c.user_low_id),(c.user_high_id))x(user_id) WHERE c.type='direct'`,
		`INSERT INTO conversation_messages(id,conversation_id,sender_user_id,client_request_id,request_fingerprint,text) SELECT gen_random_uuid(),c.id,CASE WHEN c.type='direct' THEN c.user_low_id ELSE md5('user'||(i%20+1))::uuid END,gen_random_uuid(),repeat('b',64),'Synthetic personal message '||i FROM conversations c CROSS JOIN generate_series(1,50)i`,
	}
	if err := db.Transaction(func(tx *gorm.DB) error {
		for _, q := range queries {
			if err := tx.Exec(q).Error; err != nil {
				return err
			}
		}
		return nil
	}); err != nil {
		t.Fatal(err)
	}
	if err := db.Exec("ANALYZE").Error; err != nil {
		t.Fatal(err)
	}
}
