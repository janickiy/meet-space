package sfu

import (
	"github.com/janickiy/meet-space/internal/domain/media"
	"os"
	"syscall"
)

func openFDs() int {
	for _, path := range []string{"/proc/self/fd", "/dev/fd"} {
		if entries, err := os.ReadDir(path); err == nil {
			return len(entries)
		}
	}
	return -1
}
func processCPU(r syscall.Rusage) float64 {
	return float64(r.Utime.Sec+r.Stime.Sec) + float64(r.Utime.Usec+r.Stime.Usec)/1e6
}
func ownPublication(p *peer, source media.Source) *publishedTrack {
	p.mu.Lock()
	defer p.mu.Unlock()
	for _, track := range p.publications {
		if track.metadata.Source == source {
			return track
		}
	}
	return nil
}
